\section{1. Causal light-travel geometry}

a feature at speed $\beta=v/c$ and viewing angle $\theta$ has apparent speed
\[
\beta_{\mathrm{app}} = \frac{\beta\sin\theta}{1-\beta\cos\theta}.
\]
This is a geometric mapping only and does not imply acausal transport.

\section{2. Doppler and beaming}
\[
\delta = [\Gamma(1-\beta\cos\theta)]^{-1},\qquad
S_{\nu,\mathrm{obs}} = \delta^{k-\alpha}S'_{\nu'}.
\]
With $k\approx2$ for continuous flow and $k\approx3$ for discrete blobs.

\section{3. Pattern and bulk speed}
Let $\beta_p$ denote pattern speed and $\beta_b$ bulk speed.
A robust model can maintain separate priors and fit observationally:
\[
\beta_{\mathrm{app},i}=f(\beta_p,\theta),\quad \text{and}
\quad \Gamma_b\,\text{from independent constraints}.
\]

\section{4. Radiative transfer and arrival time}
Ray-transfer equation:
\[
\frac{dI_\nu}{ds}=-\kappa_\nu I_\nu + j_\nu.
\]
Observer time map:
\[
t_\mathrm{obs}=t_\mathrm{em}+\frac{D-\mathbf n\cdot\mathbf r}{c}.
\]

\section{5. Multi-channel transport}
For each channel $c$ (CR, neutrino, EM):
\[
\frac{\partial f_c}{\partial t}+\mathbf v\cdot\nabla f_c=
\nabla\cdot(\mathbf D\nabla f_c)+\frac13(\nabla\cdot\mathbf u)
\frac{\partial f_c}{\partial\ln p}+Q_c-L_c.
\]

\section{6. Shock acceleration proxy}
Test-particle slope:
\[
s=(r+2)/(r-1),
\]
with compression ratio $r$. A simple confinement bound is
\[
E_\mathrm{max}\approx Z e B R \beta.
\]

\section{7. Event scheduling}
Event state parameters:
\[
\mathcal E=(t_0,\mathbf r_0,\theta, P(t),\phi_c).
\]
Each channel uses delayed kernel
\[
K_c(t)=H(t)\left(\frac{t}{\tau_c}\right)e^{-t/\tau_c}.
\]

\section{8. Inference objective}
\[
\mathcal L(\theta)=\sum_{\nu,t}\frac{(X_\mathrm{model}(t,\nu;\theta)-X_\mathrm{obs}(t,\nu))^2}{\sigma^2}
\]
with low-dimensional Bayesian exploration on parameters $\theta$.
